-- Create the Vehicle table
CREATE TABLE Vehicle (
    vehicle_no VARCHAR(20) PRIMARY KEY,
    owner VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    registered_date DATE NOT NULL
);
